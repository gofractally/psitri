cmake_minimum_required(VERSION 3.20)
project(psitri_api_bank LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Threads REQUIRED)
get_filename_component(PSITRI_BENCH_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(API_BANK_INTEGRATIONS "" CACHE PATH "Pinned separate integration source; empty builds the packaged MDBX pair")
set(API_BANK_ABSEIL "" CACHE PATH "Pinned Abseil source for the RocksDB shim")
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(PSITRI_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
option(API_BANK_NATIVE_ARCH "Tune for the runner CPU" ON)
if(API_BANK_NATIVE_ARCH)
    add_compile_options(-march=native)
endif()

function(api_binary name api library)
    add_executable(fair-bank-${name} ../fair_bank/main.cpp)
    target_compile_definitions(fair-bank-${name} PRIVATE FAIR_${api}=1 FAIR_API_PAIR=1)
    target_link_libraries(fair-bank-${name} PRIVATE ${library} Threads::Threads)
    set_target_properties(fair-bank-${name} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endfunction()

if(API_BANK_INTEGRATIONS)
    # Separate build prevents symbol collisions between the integration engines
    # and the maintained engines. Each pair still compiles the same API adapter.
    add_compile_definitions(PSITRI_PLATFORM_OPTIMIZATIONS=1)
    add_subdirectory("${FETCHCONTENT_SOURCE_DIR_PSIO}/cpp" psio EXCLUDE_FROM_ALL)
    foreach(library sal hash ucc art psitri)
        add_subdirectory("${API_BANK_INTEGRATIONS}/libraries/${library}" integrations/${library} EXCLUDE_FROM_ALL)
    endforeach()
    if(NOT API_BANK_ABSEIL)
        message(FATAL_ERROR "Provide pinned API_BANK_ABSEIL source")
    endif()
    set(ABSL_BUILD_TESTING OFF CACHE BOOL "" FORCE)
    set(ABSL_ENABLE_INSTALL OFF CACHE BOOL "" FORCE)
    add_subdirectory("${API_BANK_ABSEIL}" abseil EXCLUDE_FROM_ALL)
    add_subdirectory("${API_BANK_INTEGRATIONS}/libraries/psitrirocks" psitrirocks EXCLUDE_FROM_ALL)
    add_subdirectory("${API_BANK_INTEGRATIONS}/libraries/psitri-sqlite" psitri-sqlite EXCLUDE_FROM_ALL)
    execute_process(COMMAND git rev-parse HEAD WORKING_DIRECTORY "${API_BANK_INTEGRATIONS}"
                    OUTPUT_VARIABLE shim_revision OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
    find_path(FAIR_ROCKSDB_INCLUDE_DIR rocksdb/db.h REQUIRED)
    find_library(FAIR_ROCKSDB_LIBRARY NAMES rocksdb REQUIRED)
    find_package(SQLite3 REQUIRED)
    if(TARGET SQLite::SQLite3 AND NOT TARGET SQLite3::SQLite3)
        add_library(SQLite3::SQLite3 ALIAS SQLite::SQLite3)
    endif()
    api_binary(rocksdb rocksdb "${FAIR_ROCKSDB_LIBRARY}")
    target_include_directories(fair-bank-rocksdb PRIVATE "${FAIR_ROCKSDB_INCLUDE_DIR}")
    api_binary(psitrirocks rocksdb psitrirocks)
    api_binary(sqlite sqlite SQLite3::SQLite3)
    api_binary(psitri_sqlite sqlite psitri-sqlite)
    foreach(shim psitrirocks psitri_sqlite)
        target_compile_definitions(fair-bank-${shim} PRIVATE FAIR_SHIM=1 FAIR_SHIM_REVISION="${shim_revision}")
    endforeach()
    foreach(engine rocksdb psitrirocks)
        target_compile_definitions(fair-bank-${engine} PRIVATE FAIR_ROCKSDB_SYNC_LEVEL="unknown")
    endforeach()
    set(dependencies "\"integrations\": \"${API_BANK_INTEGRATIONS}\", \"abseil\": \"${API_BANK_ABSEIL}\"")
    set(libraries "\"${FAIR_ROCKSDB_LIBRARY}\", \"${SQLite3_LIBRARY}\"")
else()
    add_subdirectory("${PSITRI_BENCH_SOURCE}" psitri EXCLUDE_FROM_ALL)
    find_path(FAIR_MDBX_INCLUDE_DIR mdbx.h REQUIRED)
    find_library(FAIR_MDBX_LIBRARY NAMES mdbx REQUIRED)
    api_binary(mdbx mdbx "${FAIR_MDBX_LIBRARY}")
    target_include_directories(fair-bank-mdbx PRIVATE "${FAIR_MDBX_INCLUDE_DIR}")
    api_binary(psitrimdbx mdbx psitrimdbx)
    target_compile_definitions(fair-bank-psitrimdbx PRIVATE FAIR_SHIM=1)
    set(dependencies "\"ucc\": \"${FETCHCONTENT_SOURCE_DIR_UCC}\"")
    set(libraries "\"${FAIR_MDBX_LIBRARY}\"")
endif()
file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/fair-bank-build.json" CONTENT
"{\n  \"compiler\": \"${CMAKE_CXX_COMPILER}\",\n  \"compiler_version\": \"${CMAKE_CXX_COMPILER_VERSION}\",\n  \"build_type\": \"${CMAKE_BUILD_TYPE}\",\n  \"native_dependencies\": {\"psio\": \"${FETCHCONTENT_SOURCE_DIR_PSIO}\", ${dependencies}},\n  \"libraries\": [${libraries}]\n}\n")

cmake_minimum_required(VERSION 3.20)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(psitri_durability_contract LANGUAGES C CXX)
  set(CMAKE_CXX_STANDARD 23)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
  if(NOT PSITRI_DURABILITY_UCC_SOURCE)
    message(FATAL_ERROR "Set PSITRI_DURABILITY_UCC_SOURCE to an explicit qualified UCC source tree")
  endif()
  set(UCC_ENABLE_TESTS OFF)
  set(ADB_ENABLE_TESTS OFF)
  set(SAL_BUILD_TESTS OFF)
  set(SAL_ART_REF_BUILD_TESTS OFF)
  set(START_REF_BUILD_TESTS OFF)
  get_filename_component(psitri_source "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
  add_subdirectory("${PSITRI_DURABILITY_UCC_SOURCE}" "${CMAKE_BINARY_DIR}/dependencies/ucc")
  foreach(library adb sal start_ref sal_art_ref psitri_api)
    add_subdirectory("${psitri_source}/libraries/${library}" "${CMAKE_BINARY_DIR}/dependencies/${library}")
  endforeach()
  enable_testing()
  if(PSITRI_DURABILITY_CATCH2_SOURCE)
    set(CATCH_BUILD_TESTING OFF)
    add_subdirectory("${PSITRI_DURABILITY_CATCH2_SOURCE}" "${CMAKE_BINARY_DIR}/dependencies/Catch2")
  endif()
  option(PSITRI_DURABILITY_API_TESTS "Run the existing public API regression suite" OFF)
  if(PSITRI_DURABILITY_CATCH2_SOURCE OR PSITRI_DURABILITY_API_TESTS)
    if(NOT TARGET Catch2::Catch2WithMain)
      find_package(Catch2 3 REQUIRED)
    endif()
    add_executable(durability_api_regression ../api/conformance.cpp ../api/large_values.cpp ../api/small_large_values.cpp)
    target_link_libraries(durability_api_regression PRIVATE psitri::api Catch2::Catch2WithMain)
    target_include_directories(durability_api_regression PRIVATE
      "${psitri_source}/experimental/api/include" "${psitri_source}/libraries/delta_art/include")
    add_test(NAME durability_public_api_regression COMMAND durability_api_regression)
  endif()
endif()
add_executable(durability_fault_driver fault_driver.cpp)
target_link_libraries(durability_fault_driver PRIVATE psitri::big ${CMAKE_DL_LIBS})
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME durable_io_fail_stop COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/run_matrix.py"
  --binary $<TARGET_FILE:durability_fault_driver> --output "${CMAKE_CURRENT_BINARY_DIR}/evidence")
set_tests_properties(durable_io_fail_stop PROPERTIES TIMEOUT 180)
